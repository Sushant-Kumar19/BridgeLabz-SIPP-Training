Initialize Queue normalQueue
Initialize Queue emergencyQueue

function addCustomer(name, type):
    if type == "emergency":
        enqueue(emergencyQueue, name)
    else:
        enqueue(normalQueue, name)

function serveCustomer():
    if emergencyQueue is not empty:
        customer = dequeue(emergencyQueue)
    else if normalQueue is not empty:
        customer = dequeue(normalQueue)
    else:
        print "No customers in queue"
    print "Serving: " + customer
